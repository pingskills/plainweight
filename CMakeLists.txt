cmake_minimum_required(VERSION 3.21)
project(plainweight VERSION 0.1.0 DESCRIPTION "A small native weight log for Linux" HOMEPAGE_URL "https://github.com/pingskills/plainweight" LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
option(PLAINWEIGHT_BUILD_TESTS "Build tests" ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Sql)
qt_standard_project_setup(REQUIRES 6.5)
if(QT_KNOWN_POLICY_QTP0004)
  qt_policy(SET QTP0004 NEW)
endif()
configure_file(src/config.h.in ${CMAKE_CURRENT_BINARY_DIR}/generated/config.h @ONLY)
qt_add_library(plainweight_core STATIC src/core/weight.cpp src/core/weight.h src/database/store.cpp src/database/store.h src/services/controller.cpp src/services/controller.h)
target_include_directories(plainweight_core PUBLIC src ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(plainweight_core PUBLIC Qt6::Core Qt6::Sql)
qt_add_library(plainweight_ui STATIC src/ui/theme.cpp src/ui/theme.h)
qt_add_qml_module(plainweight_ui URI PlainWeight VERSION 1.0 OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/qml/PlainWeight QML_FILES qml/Main.qml RESOURCES resources/icons/png/256/io.github.pingskills.plainweight.png)
target_link_libraries(plainweight_ui PUBLIC plainweight_core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
qt_add_executable(plainweight src/app/main.cpp)
target_link_libraries(plainweight PRIVATE plainweight_core plainweight_ui plainweight_uiplugin)
install(TARGETS plainweight RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.github.pingskills.plainweight.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/io.github.pingskills.plainweight.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
install(FILES resources/icons/io.github.pingskills.plainweight.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
foreach(size 16 32 48 64 128 256)
  install(FILES resources/icons/png/${size}/io.github.pingskills.plainweight.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps)
endforeach()
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/plainweight)
if(PLAINWEIGHT_BUILD_TESTS)
  enable_testing()
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(tst_plainweight tests/tst_plainweight.cpp)
  target_link_libraries(tst_plainweight PRIVATE plainweight_core Qt6::Test)
  add_test(NAME tst_plainweight COMMAND tst_plainweight)
  set_tests_properties(tst_plainweight PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  qt_add_executable(tst_ui_smoke tests/tst_ui_smoke.cpp)
  target_link_libraries(tst_ui_smoke PRIVATE plainweight_core plainweight_ui plainweight_uiplugin Qt6::Test)
  add_test(NAME tst_ui_smoke COMMAND tst_ui_smoke)
  set_tests_properties(tst_ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=none;QT_QUICK_CONTROLS_STYLE=Basic;PLAINWEIGHT_NO_OMARCHY=1")
endif()
